# ==========================================
# 1. CHUYỂN ĐỔI CHỮ HOA / CHỮ THƯỜNG
# ==========================================
s = "hEllO wORLd"
print(s.capitalize())  # 'Hello world'  -> Viết hoa chữ cái đầu chuỗi, còn lại thường
print(s.title())       # 'Hello World'  -> Viết hoa chữ cái đầu mỗi từ
print(s.upper())       # 'HELLO WORLD'  -> Viết hoa toàn bộ
print(s.lower())       # 'hello world'  -> Viết thường toàn bộ
print(s.swapcase())    # 'HeLLo WorlD'  -> Đảo hoa thành thường và ngược lại
print("Straße".casefold()) # 'strasse'  -> Viết thường triệt để (dùng khi so sánh chuỗi)


# ==========================================
# 2. CĂN LỀ, ĐỘ DÀI, ĐIỀN KÝ TỰ
# ==========================================
s = "Py"
print(s.center(6, "-"))    # '--Py--'     -> Căn giữa, bù ký tự '-' cho đủ 6 chỗ
print(s.ljust(5, "."))     # 'Py...'      -> Căn trái, bù '.' bên phải
print(s.rjust(5, "."))     # '...Py'      -> Căn phải, bù '.' bên trái
print("42".zfill(5))       # '00042'      -> Thêm số 0 ở đầu cho đủ 5 ký tự
print("A\tB".expandtabs(4))# 'A   B'      -> Đổi tab thành khoảng trắng (mặc định 8)


# ==========================================
# 3. TÌM KIẾM VÀ ĐẾM
# ==========================================
s = "banana"
print(s.count("a"))          # 3          -> Đếm số lần 'a' xuất hiện
print(s.find("na"))          # 2          -> Vị trí đầu tiên (không thấy trả -1)
print(s.rfind("na"))         # 4          -> Vị trí cuối cùng (không thấy trả -1)
print(s.index("na"))         # 2          -> Giống find(), nhưng không thấy báo lỗi ValueError
print(s.rindex("na"))        # 4          -> Giống rfind(), nhưng không thấy báo lỗi ValueError
print(s.startswith("ba"))    # True       -> Kiểm tra có bắt đầu bằng 'ba' không
print(s.endswith("na"))      # True       -> Kiểm tra có kết thúc bằng 'na' không


# ==========================================
# 4. KIỂM TRA KIỂU NỘI DUNG (True / False)
# ==========================================
print("User123".isalnum())       # True   -> Chỉ gồm chữ và số (không có dấu cách/ký tự đặc biệt)
print("Python".isalpha())        # True   -> Chỉ gồm chữ cái
print("Hello!".isascii())        # True   -> Nằm trong bảng mã ASCII (tiếng Việt có dấu sẽ False)
print("123".isdecimal())         # True   -> Số thập phân cơ bản (0-9)
print("²".isdigit())             # True   -> Chữ số (nhận cả số mũ ², nhưng phân số ½ thì False)
print("½".isnumeric())           # True   -> Ký tự số (nhận cả số mũ ², phân số ½, số La Mã Ⅳ)
print("var_1".isidentifier())    # True   -> Đặt làm tên biến trong Python hợp lệ (số đầu sẽ False)
print("abc".islower())           # True   -> Tất cả đều viết thường
print("ABC".isupper())           # True   -> Tất cả đều viết hoa
print("Hello World".istitle())   # True   -> Đúng chuẩn viết hoa đầu mỗi từ
print("Hi\n".isprintable())      # False  -> Có ký tự ẩn '\n' nên không in nguyên vẹn ra 1 dòng được
print("   \t\n".isspace())       # True   -> Toàn bộ là khoảng trắng (dấu cách, tab, xuống dòng)


# ==========================================
# 5. CẮT BỎ KHOẢNG TRẮNG / KÝ TỰ THỪA
# ==========================================
s = "---Hello---"
print("  hi  ".strip())   # 'hi'          -> Mặc định cắt khoảng trắng 2 đầu
print(s.strip("-"))       # 'Hello'       -> Cắt dấu '-' ở cả 2 đầu
print(s.lstrip("-"))      # 'Hello---'    -> Chỉ cắt bên trái (đầu chuỗi)
print(s.rstrip("-"))      # '---Hello'    -> Chỉ cắt bên phải (cuối chuỗi)


# ==========================================
# 6. TÁCH VÀ GHÉP CHUỖI
# ==========================================
print("a,b,c".split(","))          # ['a', 'b', 'c']      -> Tách từ trái sang
print("a,b,c".rsplit(",", 1))      # ['a,b', 'c']         -> Tách từ phải sang (tối đa 1 lần)
print("dòng 1\ndòng 2".splitlines()) # ['dòng 1', 'dòng 2'] -> Tách theo dấu xuống dòng
print("user@gmail.com".partition("@"))  # ('user', '@', 'gmail.com') -> Chia 3 phần tại chỗ gặp đầu tiên
print("a.b.c".rpartition("."))          # ('a.b', '.', 'c')          -> Chia 3 phần tại chỗ gặp cuối cùng
print("-".join(["2026", "11", "27"]))   # '2026-11-27'    -> Nối các phần tử bằng dấu '-'


# ==========================================
# 7. THAY THẾ VÀ BIẾN ĐỔI
# ==========================================
print("I like Java".replace("Java", "Python"))  # 'I like Python'

# maketrans() + translate(): Thay thế hàng loạt ký tự cùng lúc
bang_ma = str.maketrans("aeo", "430")           # Quy ước: a->4, e->3, o->0
print("hello".translate(bang_ma))               # 'h3ll0'
